ctrl+alt+t                              #打开终端
printenv                                #与“echo $”相同，打印环境变量 
rm -rf *                                #删除此文件夹下所有文件
mkdir -p 文件夹地址                      #创建母文件夹和子文件夹

pwd                                     #打印当前工作目录
sort -n                                 #数字排序输出
sed -i.bak '/^$/d' 文件名称              #删除文件空行    
$()                                     #把括号里面命令的输出结果，捕获出来当成字符串使用
seq                                     #指定步长 seq 起点 步长 终点
magick pic.png pic.jpg                  #格式转换
magick pic.jpg -resize 50% out.jpg      #百分比缩放
magick *.jpg out.pdf                    #多张图片合并为 PDF
git revert                              #新建一条提交，反向抵消指定commit的改动
$?                                      #保存上一条命令的退出码,0成功1通用错误2参数错误或文件不存在127命令找不到130Ctrl+C中断程序
set -x                                  #打开shell调试模式，执行每一行命令前，把这条命令展开后的内容打印出来
set +x                                  #关闭调试模式
| xargs                                 #命令接收文件名/操作对象做参数
curl                                    #命令行网络客户端，支持 HTTP/HTTPS/FTP 等，用来请求接口下载文件，返回内容默认输出到stdout
*                                       #匹配任意长度任意字符
?                                       #匹配单个任意字符
[]                                      #匹配括号内任意一个字符
{}                                      #只是生成字符串，不管文件有没有
>                                       #直接清空原有内容，从头覆盖
>>                                      #不清空旧文件，把新内容加到文件末尾
<                                       #把文件内容给命令的标准输入 stdin
cmd > file 2>&1                         #普通输出、报错，全部写入file
source                                  #不需要脚本有执行权限(x)
kill                                    #请求进程自行退出
ctrl+z                                  #暂停任务
fg                                      #恢复最近任务
alias                                   #给命令起短名字，简化输入
curl                                    #命令行发网络请求、传数据的开源工具
git restore                             #用来撤销工作区 / 暂存区的修改

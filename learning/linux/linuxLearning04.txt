\                                       #取消后面字符的特殊含义
which                                   #列出程序的路径
sort -u                                 #排序的同时自动删除重复行
sed -i 's/被替换内容/替换内容/g' 文件名称 #在当前文件，把文中所有`被替换内容`替换成`替换内容`，直接修改原文件
git reset --hard                        #完全清空，工作区 + 暂存区全部覆盖为目标版本，本地修改全部丢失
find 查找路径 查找条件 -exec 命令 {} \;   #find 找到文件之后，对搜索出来的每一个文件执行指定命令
.                                       #匹配任意单个字符（不匹配换行）
*                                       #匹配它前面的字符 0 次或多次
^                                       #匹配行首，写在表达式最前面
$                                       #匹配行尾，写在表达式末尾
[]                                      #字符集：匹配括号内任意一个字符
[^ ]                                    #字符集取反：匹配不在方括号里面的任意单个字符
\                                       #转义符，取消元字符特殊作用，匹配符号本身
\{n\}                                   #限定前面字符恰好 n 次；`\{n,\}`至少 n 次；`\{n,m\}`n~m 次
>                                       #输出重定向
>>                                      #追加输出（不覆盖，追加在末尾）
<                                       #输入重定向
if ... then ... fi                      #条件判断
if[] then elif[] then else if           #条件判断支持写法
while 测试命令; do 循环体; done          #条件循环
sleep                                   #让脚本 / 终端暂停、休眠指定时间

^\d$                                    #匹配单独数字
;                                       #不管前一条成败，都执行后面
&&                                      #前一条成功才执行后面
||                                      #前一条失败才执行后面
( )                                     #创建一个子shell
echo $?                                 #保存上一条命令的退出状态码
Ctrl+Z                                  #暂停，放入后台
jobs                                    #查看任务
bg 1                                    #在后台继续发包
fg 1                                    #切回前台
Ctrl+C                                  #终止程序
source                                  #在当前Shell进程直接读取并执行文件中的命令

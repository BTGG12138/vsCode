git init                                #初始化仓库
add                                     #将文件放入暂存区
commit                                  #提交到本地仓库
status                                  #查看工作区和暂存区状态
log                                     #查看提交日志
log --graph                             #图形化查看提交日志
log -p                                  #查看提交日志和修改差异
branch                                  #创建新分支
branch -d                               #删除指定分支
switch                                  #切换指定分支
merge                                   #将目标分支合并到当前分支
revert                                  #反向提交
clone                                   #克隆仓库
fetch                                   #获取最新提交到本地
diff                                    #查看代码差异
pull                                    #合并最新提交到本地
worktree                                #同一个Git仓库，可以同时开出多个独立代码文件夹（工作区），每个文件夹固定绑定一条分支。共用一套.git仓库数据

pip                                     #Python官方包管理器
uv                                      #Rust编写的新一代Python项目&包管理器
docker                                  #操作系统级容器虚拟化工具
systemd                                 #后台总管程序
URL                                     #统一资源定位符

/new                                    #新开独立会话，主线会话存档
/clear                                  #清空当前会话，全新上下文
/status                                 #查看当前token用量、模型、工作目录、工具权限状态
/retry                                  #重新执行上一轮回答，不新增对话
/undo                                   #撤销上一轮用户+模型的对话，回退一步
/btw                                    #旁路临时提问，不污染主线上下文
/side                                   #开启旁路子对话，独立上下文，结束切回主线
/model <name>                           #切换模型
/reasoning                              #调整推理强度
/raw                                    #原始输出模式，关闭美化、多余解释，只输出代码/结果
/verbose                                #切换工具调用日志详细程度，看agent内部动作
/plan                                   #让Agent先输出结构化执行计划，再动手
/export                                 #导出完整会话记录

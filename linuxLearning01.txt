ls                                      #列出目录
ls -a                                   #列出，包括隐藏目录
ls -l                                   #列出，列表包括文件细节
ls -lh                                  #列出，列表包括文件细节，文件大小单位
cd 工作目录地址                          #切换工作目录，默认进入当前用户的目录
pwd                                     #当前工作目录地址
.                                       #当前目录
..                                      #上一级目录
~                                       #home目录
#ctrl+l                                 #清屏
mkdir 文件夹名称                         #创建文件夹
mkdir -p 文件夹名称                      #创建不存在的父目录，创建连续的目录
rmdir 文件夹名称                         #只能删除空文件夹
touch 文件名称                           #创建文件
cat 文件名称                             #查看文件内容
more 文件名称                            #翻页，查看文件内容
cp 复制文件名称 粘贴文件地址              #复制文件
cp -r 复制文件夹名称 粘贴文件夹地址       #复制文件夹
mv 移动文件名称 移动文件地址              #移动，可以用来改名，可以直接移动文件夹
rm 文件名称                              #删除
rm -f 文件名称                           #强制删除
rm -r 文件夹名称                         #删除文件夹
*                                       #作为“某”的通配符
su - root                               #开启管理员模式
which 程序名称                           #查找程序文件地址
find 开始查找的地址 -name "文件名称"      #通过文件名称查找
find 开始查找的地址 -size ±文件大小kMG    #通过文件大小查找，加减是大于或者是小于，分别是KB MB GB
grep "查找内容" 查找文件名称              #查看包含内容的行的内容
grep -n "查找内容" 查找文件名称           #找到的行数，查看包含内容的行的内容
wc 文件名称                              #统计文件的行数，单词数，字节数
wc -c 文件名称                           #统计文件字节数
wc -m 文件名称                           #统计文件字符数，汉字占三字节
wc -l 文件名称                           #统计文件行数
wc -w 文件名称                           #统计文件单词数
|                                       #管道符，将左边的输出作为右边的输入
echo "输出内容"                          #输出内容
`  `                                    #包括的内容，作为结果输入
>                                       #将左边的输出覆盖写入右边的文件
>>                                      #将左边的输出追加写入右边的文件
tail 文件名称                            #查看文件尾部十行内容
tail -n 尾部行数 文件名称                 #查看文件尾部几行内容
tail -f 文件名称                         #持续追踪文件尾部内容

vi 文件名称                              #进入编辑器
i                                       #进入输入模式
dd                                      #删除该行内容
几行dd                                  #删除向下几行内容
yy                                      #复制该行内容
几行yy                                  #复制向下几行内容
p                                       #粘贴内容
u                                       #撤销内容
/搜素内容                                #搜索内容
n                                       #向下搜索
N                                       #向上搜索
esc                                     #退出编辑模式
:w                                      #保存内容
:q                                      #退出命令模式
:q!                                     #强制退出命令模式
:wq                                     #保存内容并退出命令模式
:set paste                              #设置粘贴模式
